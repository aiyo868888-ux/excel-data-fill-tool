### 1.2 电脑端（桌面应用）

**方案选择：**

**选项 A：Electron**

- 缺点：体积大，资源占用高

**选项 B：Tauri**

- 缺点：生态相对较新

- 缺点：桌面端支持尚不完善

**推荐方案：Tauri（V2）**

### 1.4 同步方案

**方案选择：**

**选项 A：云端中转（推荐）**
- 优点：可控、安全、支持多设备
- 缺点：需要维护服务器

**选项 B：Syncthing（去中心化）**
- 优点：无需服务器，隐私好
- 缺点：配置复杂，普通用户难上手

**推荐方案：云端中转 + 增量同步**

**同步流程：**
```
手机端：
1. 监听数据变化
2. 增量打包（只传变化的部分）
3. 压缩（GZIP）
4. 加密（AES-256）
5. 上传到云端

云端：
1. 接收数据
2. 验证完整性
3. 存储到对象存储
4. 更新索引（PostgreSQL）

电脑端：
1. 定时轮询（每5分钟）
2. 检查云端更新
3. 下载 JSON 文件
4. 解析并导入本地数据库
5. 触发桌面通知
```

```kotlin
                clipboardCount = it.clipboardCount + 1

        viewModelScope.launch {

                NoteType.IDEA -> createIdeaNote()
                NoteType.INSIGHT -> createInsightNote()
                NoteType.TODO -> createTodoNote()
            }
            noteRepository.saveNote(note)
            _uiState.update { it.copy(saveSuccess = true) }
        }
    }
```

```kotlin
// NoteRepository.kt
@Singleton
class NoteRepository @Inject constructor(
    private val jsonFileStorage: JsonFileStorage,
    private val syncService: SyncService
) {

    suspend fun saveNote(note: Note) {
        // 保存到本地 JSON 文件
        jsonFileStorage.appendNote(note)

        // 上传到云端
        syncService.uploadNote(note)
    }

    suspend fun getNotesByDate(date: LocalDate): List<Note> {
        return jsonFileStorage.readNotesByDate(date)
    }

    suspend fun searchNotes(query: String): List<Note> {
        return jsonFileStorage.searchNotes(query)
    }
}

// JsonFileStorage.kt
@Singleton
class JsonFileStorage @Inject constructor(
    @ApplicationContext private val context: Context
) {

    private val notesDir = File(context.filesDir, "notes")

    suspend fun appendNote(note: Note) {
        withContext(Dispatchers.IO) {
            val today = LocalDate.now()
            val file = File(notesDir, "$today.json")

            val dailyNotes = if (file.exists()) {
                readDailyNotes(file)
            } else {
                DailyNotes(date = today.toString())
            }

            dailyNotes.data.add(note)
            writeDailyNotes(file, dailyNotes)

            // 触发同步
            syncService.notifyChange()
        }
    }

    private fun readDailyNotes(file: File): DailyNotes {
        val json = file.readText()
        return Json.decodeFromString<DailyNotes>(json)
    }

    private fun writeDailyNotes(file: File, dailyNotes: DailyNotes) {
        val json = Json.encodeToString(dailyNotes)
        file.writeText(json)
    }
}
```

**D. 同步模块（SyncModule）**

```kotlin
// SyncService.kt
@Singleton
class SyncService @Inject constructor(
    private val apiClient: SyncApiClient,
    private val jsonFileStorage: JsonFileStorage
) {

    private val syncScope = CoroutineScope(Dispatchers.IO + SupervisorJob())

    fun startSync() {
        syncScope.launch {
            while (isActive) {
                try {
                    syncPendingFiles()
                    delay(5_000) // 每5秒检查一次
                } catch (e: Exception) {
                    Timber.e(e, "Sync failed")
                }
            }
        }
    }

    private suspend fun syncPendingFiles() {
        val pendingFiles = jsonFileStorage.getPendingFiles()

        pendingFiles.forEach { file ->
            try {
                uploadFile(file)
                jsonFileStorage.markSynced(file)
            } catch (e: Exception) {
                Timber.e(e, "Failed to sync ${file.name}")
            }
        }
    }

    suspend fun uploadNote(note: Note) {
        val date = note.createdAt.toLocalDate()
        val file = File(notesDir, "$date.json")

        if (file.exists()) {
            val json = file.readText()
            apiClient.uploadNote(date.toString(), json)
        }
    }
}
```

```rust
use rusqlite::{Connection, Result};

pub fn get_notes_by_date(date: &str) -> Result<Vec<Note>> {
    let conn = Connection::open("notes.db")?;

    let mut stmt = conn.prepare(
        "SELECT id, type, content, created_at FROM notes WHERE date = ?"
    )?;

    let notes = stmt.query_map([date], |row| {
        Ok(Note {
            id: row.get(0)?,
            type: row.get(1)?,
            content: row.get(2)?,
            created_at: row.get(3)?,
        })
    })?.collect::<Result<Vec<_>, _>>()?;

    Ok(notes)
}

// sync.rs
pub async fn sync_with_cloud() -> Result<SyncStatus> {
    // 1. 获取云端文件列表
    let remote_files = api::get_file_list().await?;

    // 2. 检查本地是否有新文件需要上传
    let local_files = fs::read_dir("notes")?;

    // 3. 下载缺失的文件
    for file in remote_files {
        if !local_file_exists(&file.date) {
            let content = api::download_file(&file.date).await?;
            save_json_file(&file.date, &content)?;
        }
    }

    Ok(SyncStatus {
        last_sync: Utc::now(),
        files_synced: remote_files.len(),
    })
}
```

```typescript
export class NoteService {
  async uploadNote(userId: string, date: string, data: string) {
    // 1. 验证数据
    const parsed = JSON.parse(data);
    this.validateNoteData(parsed);

    // 2. 存储到对象存储
    const key = `${userId}/${date}.json`;
    await storageService.upload(key, data);

    // 3. 更新索引
    await prisma.note.upsert({
      where: { userId_date: { userId, date } },
      update: { updatedAt: new Date() },
      create: { userId, date, size: data.length },
    });

    return { success: true };
  }

  async getNoteList(userId: string) {
    const notes = await prisma.note.findMany({
      where: { userId },
      orderBy: { date: 'desc' },
    });

    return notes.map((note) => ({
      date: note.date,
      updated_at: note.updatedAt,
      size: note.size,
    }));
  }

  async downloadNote(userId: string, date: string) {
    const key = `${userId}/${date}.json`;
    return await storageService.download(key);
  }
}
```

### 4.2 数据加密

**手机端加密：**
```kotlin
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.spec.IvParameterSpec

class EncryptionManager {
    private val key: SecretKey = generateKey()

    fun encrypt(data: String): String {
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        cipher.init(Cipher.ENCRYPT_MODE, key)
        val encrypted = cipher.doFinal(data.toByteArray())
        return Base64.encodeToString(encrypted, Base64.DEFAULT)
    }

    fun decrypt(data: String): String {
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        cipher.init(Cipher.DECRYPT_MODE, key, IvParameterSpec(ByteArray(12)))
        val decrypted = cipher.doFinal(Base64.decode(data, Base64.DEFAULT))
        return String(decrypted)
    }
}
```
